% Begin the first row of side-by-side sections (Body and Mind)
\begin{multicols}{2}
\section{Body}
\begin{itemize}[leftmargin=*]
\item Eat on time, simple and nutritious.
\item Sleep enough, at fixed times.
\item Move the body: run, stretch, lift.
\item Maintain cleanliness and silence.
\end{itemize}

\columnbreak % Forces "Mind" into the right column

\section{Mind}
\begin{itemize}[leftmargin=*]
\item Study Mathematics, solve problems.
\item Read literature, especially philosophy.
\item Protect study time; it's sacred.
\item Seek coherence and elegance.
\end{itemize}
\end{multicols}

% Begin the second row of side-by-side sections (Profession and Money)
\begin{multicols}{2}
\section{Profession}
\begin{itemize}[leftmargin=*]
\item Do small, precise work every day.
\item Avoid multitasking and drama.
\item Keep tools simple and durable.
\item Build minimal, permanent systems.
\end{itemize}

\columnbreak % Forces "Money" into the right column

\section{Money}
\begin{itemize}[leftmargin=*]
\item Save automatically; spend consciously.
\item Eliminate waste wherever possible.
\item Plan monthly and review weekly.
\item Treat time as currency.
\end{itemize}
\end{multicols}

% This section spans the full width at the bottom
\section{Relationships}
\begin{itemize}[leftmargin=*]
\item Give respect, not indulgence.
\item Avoid arguments that prove nothing.
\item Keep promises; apologize when wrong.
\item Prefer kindness to cleverness.
\end{itemize}
